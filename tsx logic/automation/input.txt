  fps: 30,
  width: 1920,
  height: 1080,
};

  return x - Math.floor(x);

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',

  // Calendar count-up 0 -> 90 days
  const dayCount = Math.floor(interpolate(frame, [20, 140], [0, 90], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic) }));
  const alertPulse = 1 + Math.sin(frame / 6) * 0.06;
  const camZoom = interpolate(frame, [0, durationInFrames], [1, 1.06], { extrapolateRight: 'clamp' });

    <svg width="120" height="120" viewBox="0 0 24 24" fill="none">
