import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01_AgeRange',
  durationInSeconds: 30,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01_AgeRange: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Counter animation from 15 to 29
  const ageCounter = Math.floor(interpolate(frame, [30, 90], [15, 29], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }));
  const cameraZoom = interpolate(frame, [0, durationInFrames], [1, 1.05], { extrapolateRight: 'clamp' });

  // Floating particles
  const particles = Array.from({ length: 25 }, (_, i) => ({
    x: seededRandom(i) * 1920,
    y: seededRandom(i + 100) * 1080,
    size: seededRandom(i + 200) * 6 + 3,
    speed: seededRandom(i + 300) * 0.5 + 0.2,
  }));

  return (
    <AbsoluteFill style={{ backgroundColor: '#76ABAE', overflow: 'hidden', opacity: exitOp }}>
      {/* Dynamic Gradient Background Layer */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle at 30% 40%, rgba(63,79,68,0.35) 0%, rgba(118,171,174,0) 60%)',
      }} />

      {/* Animated Shape Layers - drifting circles & rectangles */}
      <div style={{
        position: 'absolute', top: 100 + Math.sin(frame * 0.02) * 20, left: 1500,
        width: 400, height: 400, borderRadius: '50%',
        border: '3px solid #3F4F44', opacity: 0.06,
        transform: `rotate(${frame * 0.2}deg) scale(${1 + Math.sin(frame * 0.03) * 0.05})`,
      }} />
      <div style={{
        position: 'absolute', top: 600, left: -100,
        width: 500, height: 500, borderRadius: 40,
        backgroundColor: '#3F4F44', opacity: 0.05,
        transform: `rotate(${frame * -0.15}deg)`,
      }} />
      <div style={{
        position: 'absolute', top: 750, left: 1300,
        width: 250, height: 250,
        backgroundColor: '#EEEEEE', opacity: 0.04,
        transform: `rotate(${45 + frame * 0.1}deg) scale(${1 + Math.cos(frame * 0.025) * 0.08})`,
      }} />

      {/* Motion Texture — vertical lines grid */}
      <svg style={{ position: 'absolute', inset: 0, opacity: 0.08 }} width="1920" height="1080">
        {Array.from({ length: 20 }).map((_, i) => (
          <line key={i} x1={i * 100 + (frame * 0.5) % 100} y1={0}
            x2={i * 100 + (frame * 0.5) % 100} y2={1080}
            stroke="#3F4F44" strokeWidth="1" />
        ))}
      </svg>

      {/* Floating Particles */}
      {particles.map((p, i) => (
        <div key={i} style={{
          position: 'absolute',
          left: p.x + Math.sin((frame + i * 20) * 0.02) * 30,
          top: (p.y + frame * p.speed) % 1080,
          width: p.size, height: p.size, borderRadius: '50%',
          backgroundColor: '#EEEEEE', opacity: 0.35,
        }} />
      ))}

      {/* Main Content */}
      <AbsoluteFill style={{ transform: `scale(${cameraZoom})`, justifyContent: 'center', alignItems: 'center' }}>
        {/* Top Label */}
        <div style={{
          fontFamily: 'Inter, Poppins',
          fontSize: 42, fontWeight: 700, letterSpacing: 8,
          color: '#3F4F44',
          opacity: sp(5),
          transform: `translateY(${(1 - sp(5)) * -30}px)`,
          marginBottom: 40,
          textTransform: 'uppercase',
        }}>
          ▎ यह वीडियो सिर्फ तेरे लिए है
        </div>

        {/* Hero Number Counter */}
        <div style={{
          display: 'flex', alignItems: 'center', gap: 60,
          transform: `scale(${sp(15)})`,
        }}>
          <div style={{
            fontFamily: 'Yantra One',
            fontSize: 340, fontWeight: 900, lineHeight: 1,
            color: '#EEEEEE',
            textShadow: '8px 8px 0px #3F4F44',
          }}>
            {ageCounter}
          </div>
          <div style={{
            fontFamily: 'Inter',
            fontSize: 180, fontWeight: 900,
            color: '#3F4F44',
            opacity: sp(25),
          }}>
            —
          </div>
          <div style={{
            fontFamily: 'Yantra One',
