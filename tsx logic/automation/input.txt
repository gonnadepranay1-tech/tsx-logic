  fps: 30,

  height: 1080,
};

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',

    const y = yBase + Math.sin(frame * 0.03 + i) * 40;
    const size = 3 + seededRandom(i * 2.2) * 6;
    const op = 0.15 + seededRandom(i) * 0.25;
    return { x, y, size, op };

    <AbsoluteFill style={{ backgroundColor: '#FF6600', overflow: 'hidden', opacity: exitOp }}>

      <AbsoluteFill style={{
