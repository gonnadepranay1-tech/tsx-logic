        {/* steps */}

        </div>

        {/* closing warning */}
        <div style={{

          transform: `translateY(${interpolate(sp(150), [0, 1], [40, 0])}px)`,

          fontFamily: 'Inter', fontSize: 48, fontWeight: 700, color: '#8B1A00',

      {/* Vignette overlay */}

      {/* Watermark */}
      <div style={{ position: 'absolute', bottom: 30, right: 45, fontFamily: 'Inter', fontSize: 22, fontWeight: 600, color: '#1D2128', opacity: 0.35 }}>
        KuxAI.7x
      </div>

  );
};
